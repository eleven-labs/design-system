import { Meta } from '@storybook/addon-docs/blocks';
import { Layout, Table } from '@/storybook/components';
import { tokenVariables } from '@/tokens';

<Meta
    title="Foundations/Design Tokens/Typography/FontFamily"
    parameters={{
        viewMode: 'docs',
        previewTabs: {
            canvas: { hidden: true }
        },
        backgrounds: {
            disable: true,
            grid: {
                disable: true,
            }
        },
    }}
/>

<Layout
    title="Typography"
    description="Font Family token scales"
>
    <Table
        title="Font Family"
        columns={[
            {
                name: 'name',
                label: 'Name',
            },
            {
                name: 'preview',
                label: 'Preview',
            },
            {
                name: 'weight_name',
                label: 'Weight Name',
            },
            {
                name: 'weight_value',
                label: 'Weight Value',
            }
        ]}
        rows={Object.entries(tokenVariables["font-family"]).reduce((rows, [tokenName, token]) => {
            const rowsByWeight = token.weights.map(weight => ({
                name: (
                    <span>{tokenName} - {weight}</span>
                ),
                preview: (
                    <span style={{ fontFamily: token.value, fontWeight: tokenVariables['font-weight'][weight].value }}>
                        {token.name} {weight}
                    </span>
                ),
                weight_name: (
                    <span>{weight}</span>
                ),
                weight_value: (
                    <span>{tokenVariables['font-weight'][weight].value}</span>
                ),
            }));

          return [...rows, ...rowsByWeight];
        }, [])}
    />
</Layout>
